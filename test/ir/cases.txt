function first[T](items: T[]): Option[T] {
    if (items.length() == 0) { return None }
    return Some(items[0])
}
print(first([1]), first(['a']), first([1.5]))

type Pair[A, B] = { a: A, b: B }
function Pair.swap(self): Pair[B, A] {
    return { a: self.b, b: self.a }
}
immut p: Pair[int, string] = { a: 1, b: 'x' }
print(p.swap())

type Conn = { id: int }
function Conn.drop(mut self) {
    print('closed', self.id)
}
function open(id: int): Conn {
    return { id: id }
}
immut a = open(1)
mut b = open(2)
b = open(3)
immut cs: Conn[] = [open(4)]
