---- file counter.kj
export opaque type Counter = {
    name: string
    count: int
}

export opaque type Shape = {
    Circle(radius: int)
    Square(side: int)
}

export function counter(name: string): Counter {
    return { name: name, count: 0 }
}

function Counter.display(self): string {
    return `<counter ${self.name}=${self.count}>`
}

function Counter.add(mut self) {
    self.count += 1
}

function Counter.total(self): int {
    return self.count
}

export function circle(radius: int): Shape {
    return Circle(radius)
}

function Shape.area(self): int {
    return match (self) {
        Circle(radius) => 3 * radius * radius
        Square(side) => side * side
    }
}

12 [<Shape>] <Shape>

==== stdlib_file
---- source
import file

match (file.open('notes.txt', file.Write)) {
    Ok(f) => {
        print(f.write('Hello world!'))
        f.close()
    }
    Err(error) => print(error)
}

match (file.open('notes.txt', file.Read)) {
    Ok(f) => {
        match (f.read()) {
            Ok(content) => print(content)
            Err(error) => print(error)
        }
        f.close()
    }
    Err(error) => print(error)
}

match (file.open('/nonexistent/notes.txt', file.Read)) {
    Ok(_) => print('opened?')
    Err(NotFound(missing)) => print(`not found: ${missing}`)
    Err(error) => print(error)
}

import { Mode } from 'file'

match (file.open('notes.txt', Append)) {

    Err(error) => Err(error)

    Ok(f) => print(f.read())

export type Point = {
    x: int
    y: int
}

function Point.origin(): Point {
    return { x: 0, y: 0 }
}

function Point.sum(self): int {
    return self.x + self.y
}

export function make(x: int, y: int): Point {
    return { x: x, y: y }
}

export function double(mut n: int) {
    n = n * 2
}

==== files_are_closed_by_their_owner
---- main main.kj
---- source
import file

function save(path: string): Result[int, file.FileError] {
    mut f = file.open(path, file.Write)?

    f.write('one ')?

    immut g = f

    g.write('two')?

    return Ok(2)
}

function load(path: string): Result[string, file.FileError] {
    immut f = file.open(path, file.Read)?

    return f.read()
}

print(save('notes.txt'))
print(load('notes.txt'))

immut h = file.open('notes.txt', file.Read)?

h.close()
---- output
Ok(2)
Ok('one two')

export type Box[T] = {
    value: T
}

function Box.get(self): T {
    return self.value
}

}
