type Person = {
name: string
age: int
tags: string[]
}

type Tree = {
Leaf
Node(left: Tree, value: int, right: Tree)
}

type AppError = {
Parse(message: string)
Range(value: int)
}

function insert(tree: Tree, value: int): Tree {
return match (tree) {
Leaf => Node(Leaf, value, Leaf)
Node(left, v, right) => if (value < v) { Node(insert(left, value), v, right) } else { Node(left, v, insert(right, value)) }
}
}

function sum(tree: Tree): int {
match (tree) {
Leaf => return 0
Node(left, value, right) => return sum(left) + value + sum(right)
}
}

function check(n: int): Result[int, string] {
if (n < 0) { return Err('negative') }

return Ok(n * 2)
}

function twice(n: int): Result[int, AppError] {
immut a = check(n).mapErr(Parse)?
return Ok(a * 2)
}

function triple(n: int): Result[int, AppError] {
immut a = check(n)?
return Ok(a * 3)
}

function first(items: int[]): Option[int] {
if (items.length() == 0) { return None }

return Some(items[0])
}

function addFirst(items: int[]): Option[int] {
immut value = first(items)?
return Some(value + 1)
}
