type P = { name: string }

mut saved: P[] = []

function same(p: P): P {
    return p
}

function later(p: P): () => string {
    return () => p.name
}

function store(p: P) {
    saved.push(p)
}

function relay(p: P) {
    store(p)
}

function depth(p: P, n: int): int {
    if (n == 0) { return p.name.length() }

    immut inner: P = { name: p.name + '!' }
    return depth(inner, n - 1)
}

function run(): string[] {
    immut a: P = { name: 'a' }
    immut b = same(a)
    immut c: P = { name: 'c' }
    immut f = later(c)
    immut d: P = { name: 'd' }
    relay(d)
    immut e: P = { name: 'e' }
    return [b.name, f(), string(depth(e, 3))]
}

function largest[T: orderable](items: T[]): Option[T] {
if (items.length() == 0) { return None }

mut best = items[0]

if (item > best) { best = item }
}

}

return total

type Box = {
    items: int[]
}

function mark(mut held: boolean[], which: int[]) {
    for local in which {
        held[local] = true
    }

    for local in which {
        held[local + 1] = false
    }
}

mark(mut held, [1, 3])
